cmake_minimum_required(VERSION 3.16)
project(fisheye_stereo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.5 REQUIRED COMPONENTS core calib3d imgproc imgcodecs)
find_package(Threads REQUIRED)
add_library(stereo_calibration src/calibration.cpp)
target_include_directories(stereo_calibration PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(stereo_calibration PUBLIC ${OpenCV_LIBS})
add_executable(fisheye_stereo src/main.cpp)
target_link_libraries(fisheye_stereo PRIVATE stereo_calibration)
add_executable(fisheye_server src/server.cpp)
target_include_directories(fisheye_server PRIVATE third_party)
target_link_libraries(fisheye_server PRIVATE stereo_calibration Threads::Threads)
enable_testing()
add_executable(stereo_test tests/synthetic.cpp)
target_link_libraries(stereo_test PRIVATE stereo_calibration)
add_test(NAME synthetic_geometry COMMAND stereo_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_and_http COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
    $<TARGET_FILE:fisheye_stereo> $<TARGET_FILE:fisheye_server> $<TARGET_FILE:stereo_test>)
endif()
